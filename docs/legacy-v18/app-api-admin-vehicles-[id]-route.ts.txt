import { eq } from "drizzle-orm";
import { getDb } from "@/db";
import { vehicleMedia, vehicles } from "@/db/schema";
import { getStudioUser } from "@/lib/studio-auth";
import { vehicleInput } from "@/lib/vehicle-input";

export const dynamic = "force-dynamic";

export async function PATCH(request: Request, { params }: { params: Promise<{ id: string }> }) {
  if (!await getStudioUser()) return Response.json({ error: "Unauthorized" }, { status: 401 });
  const { id } = await params;
  const parsed = vehicleInput.safeParse(await request.json().catch(() => null));
  if (!parsed.success) return Response.json({ error: "Check the vehicle details." }, { status: 400 });
  const db = getDb();
  const existing = await db.select({ id: vehicles.id }).from(vehicles).where(eq(vehicles.id, id)).limit(1);
  if (!existing.length) return Response.json({ error: "Vehicle not found." }, { status: 404 });
  if (parsed.data.status === "published") {
    const v = parsed.data;
    const images = await db.select({ id: vehicleMedia.id }).from(vehicleMedia).where(eq(vehicleMedia.vehicleId, id)).limit(1);
    if (!v.year || v.priceRm == null || v.mileageKm == null || !images.length) return Response.json({ error: "Add the year, price, mileage and at least one photo before publishing." }, { status: 400 });
  }
  try {
    await db.update(vehicles).set({ ...parsed.data, updatedAt: new Date().toISOString() }).where(eq(vehicles.id, id));
  } catch {
    return Response.json({ error: "Stock ID already exists or could not be saved." }, { status: 409 });
  }
  return Response.json({ ok: true });
}
